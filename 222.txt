// №1
function func1(localNum) {
	console.log(localNum); // 1
}
func1(1);


// №2
function func2(localNum) {
	console.log(localNum); // 1
}
let num2 = 1;
func2(num2);


// №3
function func3(localNum) {
	console.log(localNum); // 1
}
let num3 = 1;
func3(num3);
num3 = 2; // меняем num ПОСЛЕ вызова функции - на вывод уже не влияет


// №4
let num4 = 1;
function func4(localNum) {
	console.log(localNum); // 2
}
num4 = 2; // меняем num ДО вызова функции
func4(num4);


// №5
function func5(localNum) {
	localNum = 2; // меняется только локальная копия
}
let num5 = 1;
func5(num5);
console.log(num5); // 1


// №6
function func6(localNum) {
	num6 = 2; // меняем глобальную num (нет let/const внутри функции)
}
let num6 = 1;
func6(num6);
console.log(num6); // 2


// №7
function func7(localNum) {
	let num7 = 2; // создаём НОВУЮ локальную переменную num
}
let num7 = 1;
func7(num7);
console.log(num7); // 1
